// controllers/chatController.js (only sendMessage changed)
const chatService = require('../services/chatservice');
const messageService = require('../services/messageService');
const orchestratorService = require('../services/orchestratorService');
const vectorService = require('../services/vectorService');

class ChatController {
  // ... other methods unchanged

  async sendMessage(req, res, next) {
    try {
      const { sessionId, customerId, content, messageType } = req.body;

      if (!sessionId || !customerId || !content) {
        return res.status(400).json({
          success: false,
          error: 'Session ID, customer ID, and content are required'
        });
      }

      const session = await chatService.getSessionById(sessionId);
      if (!session || session.customer_id !== customerId) {
        return res.status(404).json({
          success: false,
          error: 'Session not found or access denied'
        });
      }

      // 1) Save user message
      const userMessage = await messageService.saveMessage(sessionId, content, 'customer', messageType);

      // 2) Upsert user message embedding
      vectorService.upsertMessageEmbedding({
        sessionId,
        messageId: userMessage.id,
        text: content,
        role: 'customer'
      });

      // 3) Orchestrate AI
      const { aiText } = await orchestratorService.respondToMessage({
        sessionId,
        customerId,
        content
      });

      // 4) Save AI message
      const aiMessage = await messageService.saveMessage(sessionId, aiText, 'ai', 'text');

      // 5) Upsert AI message embedding
      vectorService.upsertMessageEmbedding({
        sessionId,
        messageId: aiMessage.id,
        text: aiText,
        role: 'ai'
      });

      res.status(200).json({
        success: true,
        data: { userMessage, aiMessage }
      });
    } catch (error) {
      next(error);
    }
  }
}

module.exports = new ChatController();
